1.1 Pytorch

1.4 Dataset

使用CIFAR10数据集来简单介绍，CIFAR10是一个图片分类数据集

1
2
3
4
5
6
7

data_set_transform = torchvision.transforms.Compose([
    torchvision.transforms.PILToTensor()
])

# 区别在于train参数是否为True，分别对应于训练数据集和测试数据集
train_set = torchvision.datasets.CIFAR10(root='./data', train=True, transform=data_set_transform, download=True)
test_set = torchvision.datasets.CIFAR10(root='./data', train=False, transform=data_set_transform, download=True)

这里我们首先定义了一个transform，用于将数据集中的PILimage类型转换为tensor类型

  • root参数为数据集下载或者加载地址
  • train参数区分此数据集为训练或测试数据集
  • transform用于将数据集中数据元素转换为对应类型
  • download用于指定如果数据集为下载是否自动下载

这样我们就将数据集下载到了本地

class Net(nn.Module):
    def __init__(self):
        super(Net, self).__init__()
        self.net = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, stride=1, padding=2),
            nn.MaxPool2d(2),
            nn.Conv2d(32, 32, 5, 1, 2),
            nn.MaxPool2d(2),
            nn.Conv2d(32, 64, kernel_size=5, padding=2),
            nn.MaxPool2d(2),
            nn.Flatten(),
            nn.Linear(1024, 64),
            nn.Linear(64, 10)
        )

    def forward(self, x):
        x = self.net(x)
        return x

epoch = 10

for i in range(epoch):
    print('Epoch {}/{}'.format(i + 1, epoch))
    for data in train_dataloader:
        imgs, tragets = data
        outputs = net(imgs)
        loss = lost_func(outputs, tragets)
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        total_train_step += 1
        if total_train_step % 100 == 0:
            print("train step {}, loss {}".format(total_train_step, loss.item()))
            writer.add_scalar('train_loss', loss.item(), total_train_step)

    # 测试步骤
    total_test_loss = 0
    total_accuracy = 0
    with torch.no_grad():
        for data in test_dataloader:
            imgs, tragets = data
            outputs = net(imgs)
            loss = lost_func(outputs, tragets)
            total_test_step += 1
            total_test_loss += loss.item()
            # 计算数据精度
            accuracy = (outputs.argmax(1) == tragets).sum()
            total_accuracy += accuracy.item()
    print("test loss {}".format(total_test_loss))
    print("accuracy:{}".format(total_accuracy / test_data_size))
    writer.add_scalar('test_loss', total_test_loss, total_test_step)

writer.close()
